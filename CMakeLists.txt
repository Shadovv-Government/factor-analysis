cmake_minimum_required(VERSION 3.20)
project(Portfolio VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(core src/core.cpp src/cashflow.cpp src/valuation.cpp src/factors.cpp)
add_executable(valuation_tests tests/valuation_tests.cpp)
target_link_libraries(valuation_tests PRIVATE core)
add_test(NAME valuation COMMAND valuation_tests)
add_executable(factors_tests tests/factors_tests.cpp)
target_link_libraries(factors_tests PRIVATE core)
add_test(NAME factors COMMAND factors_tests)
target_include_directories(core PUBLIC include)
if(MSVC)
 target_compile_options(core PRIVATE /W4 /utf-8)
endif()
add_executable(portfolio src/main.cpp)
target_link_libraries(portfolio PRIVATE core)
enable_testing()
add_executable(tests tests/tests.cpp)
target_link_libraries(tests PRIVATE core)
add_test(NAME core_tests COMMAND tests)

add_executable(dashboard src/dashboard.cpp)
if(MSVC)
 target_compile_options(dashboard PRIVATE /W4 /utf-8)
endif()
add_test(NAME cf_integration COMMAND dashboard "${CMAKE_SOURCE_DIR}/tests/fixtures/cf" unused.html --check)

add_executable(portfolio_app src/app.cpp)
target_link_libraries(portfolio_app PRIVATE core ws2_32 shell32)
if(MSVC)
 target_compile_options(portfolio_app PRIVATE /W4 /utf-8)
endif()

add_dependencies(portfolio_app dashboard)

add_executable(golden_tests tests/golden_tests.cpp)
target_link_libraries(golden_tests PRIVATE core)
add_test(NAME golden_seed42 COMMAND golden_tests golden)
add_test(NAME profiles_match_code COMMAND golden_tests profiles profiles WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
add_test(NAME config_errors COMMAND golden_tests errors)

add_executable(strict_tests tests/strict_tests.cpp)
target_link_libraries(strict_tests PRIVATE core)
add_test(NAME strict_calculation COMMAND strict_tests)
